# A consumer of an installed Ostia (RFC-0001 §3.2): find_package(ostia COMPONENTS ...).
cmake_minimum_required(VERSION 4.1)
project(ostia_consumer LANGUAGES C CXX)
if("${WANT}" STREQUAL "")
  find_package(ostia REQUIRED)
  if(NOT TARGET ostia::telemetry OR NOT TARGET ostia::fabric)
    message(FATAL_ERROR "find_package(ostia) without COMPONENTS must load every real component")
  endif()
else()
  find_package(ostia REQUIRED COMPONENTS ${WANT})
endif()
# Load the staged libraries, not a copy py-dev installed into the pixi environment: move
# conda's rpath behind the stage (as the main build does, RFC-0001 §3.5).
if(DEFINED OSTIA_SOURCE_DIR)
  include(${OSTIA_SOURCE_DIR}/cmake/OstiaToolchain.cmake)
  ostia_strip_conda_rpath()
endif()
if(DEFINED ENV{CONDA_PREFIX} AND DEFINED ostia_DIR)
  cmake_path(GET ostia_DIR PARENT_PATH _cmake_dir)
  cmake_path(GET _cmake_dir PARENT_PATH _stage_lib)
  set(CMAKE_BUILD_RPATH "${_stage_lib};$ENV{CONDA_PREFIX}/lib")
endif()

if("fabric" IN_LIST WANT)
  add_executable(consumer main.cpp)
  target_link_libraries(consumer PRIVATE ostia::fabric)
  target_compile_definitions(consumer PRIVATE OSTIA_EXPECT_LEVEL=${ostia_TELEMETRY_LEVEL})
endif()
